Sehatra voalohany
<!-- 1. Ampidiro ny engine -->
<script src="rakitrakatra.min.js"></script>

<!-- 2. Boribory hihazona ny lalao -->
<canvas id="gameCanvas" width="960" height="540"></canvas>

<script>
  // 3. Manomboka ny lalao — Vondrona (ECS), Kamera, Fanindry, Feo... vonona daholo
  const R = window.Rakitrakatra;
  const game = new R.Lalao(960, 540, {
    canvas: document.getElementById('gameCanvas')
  });
</script>
class SehatroKo extends R.Sehatra {
  create() {
    // entity ao amin'ny Vondrona (ECS)
    this.id = this.game.ecs.create();
    this.game.ecs.x[this.id] = 100;
    this.game.ecs.y[this.id] = 100;

    // kamera manaraka
    this.game.camera.follow({ x: 100, y: 100, w: 32, h: 32 });
  }

  update(dt) {
    const axis = R.Fanindry.axis();
    this.game.ecs.x[this.id] += axis.x * 200 * dt;
  }

  render(renderer) {
    renderer.drawRect(
      this.game.ecs.x[this.id],
      this.game.ecs.y[this.id],
      32, 32, 0xFF1493FF
    );
  }
}

game.addScene('lalao', SehatroKo);
game.start('lalao');
