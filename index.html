// ==========================================
// DEMO 2: Plugin Fizika Arcade
// ==========================================

const lalao2 = new Rakitrakatra2.Lalao(600, 350, {
    canvas: document.getElementById('canvas2')
});

const baolina = [];

lalao2.sehatra('fizika', ({ fizika }) => ({
    famoronana() {
        // Ampidiro tany (static body)
        fizika.ampio(new Rakitrakatra2.Vatana(0, 330, 600, 20, {
            static: true
        }));
        
        // Ampidiro rindrina (static bodies)
        fizika.ampio(new Rakitrakatra2.Vatana(0, 0, 20, 350, { static: true }));
        fizika.ampio(new Rakitrakatra2.Vatana(580, 0, 20, 350, { static: true }));
        
        // Ampidiro baolina maromaro
        for (let i = 0; i < 5; i++) {
            const b = new Rakitrakatra2.Vatana(
                100 + i * 80, 50 + i * 30, 25, 25, {
                    bounce: 0.6 + i * 0.05,
                    gravity: 0.5 + i * 0.1
                }
            );
            fizika.ampio(b);
            baolina.push({ body: b, color: ['#ff1493','#00ffff','#ffd700','#00ff00','#ff6600'][i] });
        }
    },
    fanavaozana(dt) {
        fizika.havaozy(dt);
    },
    sariho(ctx) {
        // Sariho tany
        ctx.fillStyle = '#2d5a27';
        ctx.fillRect(0, 330, 600, 20);
        ctx.fillStyle = '#5a4327';
        ctx.fillRect(0, 0, 20, 350);
        ctx.fillRect(580, 0, 20, 350);
        
        // Sariho baolina
        baolina.forEach(b => {
            ctx.fillStyle = b.color;
            ctx.beginPath();
            ctx.arc(b.body.x + 12.5, b.body.y + 12.5, 12.5, 0, Math.PI * 2);
            ctx.fill();
            ctx.strokeStyle = '#fff';
            ctx.lineWidth = 1;
            ctx.stroke();
        });
    }
}));

lalao2.atombohy('fizika');

// Fonction fanampiana baolina vaovao
function ampidiroBaolina() {
    const loko = ['#ff1493','#00ffff','#ffd700','#00ff00','#ff6600','#8a2be2'];
    const col = loko[Math.floor(Math.random() * loko.length)];
    const b = new Rakitrakatra2.Vatana(
        Math.random() * 500 + 50, 20, 25, 25, {
            bounce: Math.random() * 0.5 + 0.3,
            gravity: Math.random() * 0.4 + 0.3
        }
    );
    lalao2.fizika.ampio(b);
    baolina.push({ body: b, color: col });
}

// ==========================================
// DEMO 3: Plugin Vovoka (Particules)
// ==========================================

const lalao3 = new Rakitrakatra2.Lalao(600, 350, {
    canvas: document.getElementById('canvas3')
});

let vovoka;

lalao3.sehatra('particules', ({ particles }) => ({
    famoronana() {
        vovoka = particles;
        
        // Mamorona mpanaparitaka foibe (feu central)
        vovoka.emitter(300, 250, {
            rate: 8,
            angle: -Math.PI / 2,
            spread: 0.3,
            minS: 1,
            maxS: 3,
            cols: ['#ff8c00', '#ff4500', '#ffd700'],
            size: 4,
            grav: -0.05,
            life: Infinity
        });
        
        // Mamorona mpanaparitaka faharoa (boribory)
        vovoka.emitter(150, 150, {
            rate: 5,
            spread: Math.PI * 2,
            minS: 0.5,
            maxS: 2,
            cols: ['#00ffff', '#0088ff'],
            size: 3,
            grav: 0.02,
            life: Infinity
        });
    },
    fanavaozana(dt) {
        vovoka.havaozy(dt);
        
        // Mamorona fipoahana kisendrasendra
        if (Math.random() < 0.02) {
            vovoka.mipoaka(
                Math.random() * 600,
                Math.random() * 350,
                { n: 30, cols: ['#ff1493', '#ffd700', '#fff'] }
            );
        }
    },
    sariho(ctx) {
        // Sariho lafika mainty
        ctx.fillStyle = '#0a0a1a';
        ctx.fillRect(0, 0, 600, 350);
        
        // Sariho particules
        vovoka.sariho(ctx);
    }
}));

lalao3.atombohy('particules');

// Fonction fipoahana
function fipoahana() {
    for (let i = 0; i < 5; i++) {
        vovoka.mipoaka(
            Math.random() * 600,
            Math.random() * 350,
            { n: 50, cols: ['#ff1493', '#00ffff', '#ffd700', '#fff'], spread: Math.PI * 2 }
        );
    }
}

// ==========================================
// DEMO 5: Plugin Fanindry (Input)
// ==========================================
// Ampiasao: Zana-tsipìka, ZQSD/WASD, na Tsorakazo
// Space = Mitsambikina | Kitiho ny totozy = Mipoaka

const lalao5 = new Rakitrakatra2.Lalao(600, 350, {
    canvas: document.getElementById('canvas5')
});

let mpilalao, vovoka5, potipotika;

lalao5.sehatra('input', ({ fizika, particles }) => ({
    famoronana() {
        vovoka5 = particles;
        mpilalao = {
            x: 280, y: 250,
            sakany: 40, haavony: 40,
            vx: 0, vy: 0,
            aminTany: false
        };
        potipotika = [];
        
        // Ampidiro tany
        fizika.ampio(new Rakitrakatra2.Vatana(0, 330, 600, 20, { static: true }));
    },
    fanavaozana(dt) {
        const F = Rakitrakatra2.Fanindry;
        const teza = F.teza();
        
        // Hetsika
        mpilalao.vx = teza.x * 5;
        mpilalao.x += mpilalao.vx;
        
        // Fitsambikina
        if (F.justPressed(' ') && mpilalao.aminTany) {
            mpilalao.vy = -10;
            mpilalao.aminTany = false;
            Rakitrakatra2.Feo.sfx('jump');
        }
        
        // Gravity
        mpilalao.vy += 0.5;
        mpilalao.y += mpilalao.vy;
        
        // Famerana
        if (mpilalao.y >= 290) {
            mpilalao.y = 290;
            mpilalao.vy = 0;
            mpilalao.aminTany = true;
        }
        mpilalao.x = Math.max(0, Math.min(560, mpilalao.x));
        
        // Kitiho totozy = mipoaka
        if (F.mouse.justDown) {
            const pos = lalao5.kamera.toWorld(F.mouse.x, F.mouse.y);
            vovoka5.mipoaka(pos.x, pos.y, {
                n: 30,
                cols: ['#ff1493', '#00ffff', '#ffd700'],
                spread: Math.PI * 2
            });
            Rakitrakatra2.Feo.sfx('explode');
        }
        
        vovoka5.havaozy(dt);
    },
    sariho(ctx) {
        ctx.fillStyle = '#0a0a1a';
        ctx.fillRect(0, 0, 600, 350);
        ctx.fillStyle = '#2d5a27';
        ctx.fillRect(0, 330, 600, 20);
        
        // Sariho mpilalao
        ctx.fillStyle = '#00ffff';
        ctx.fillRect(mpilalao.x, mpilalao.y, mpilalao.sakany, mpilalao.haavony);
        ctx.strokeStyle = '#fff';
        ctx.strokeRect(mpilalao.x, mpilalao.y, mpilalao.sakany, mpilalao.haavony);
        
        // Sariho toromarika
        ctx.fillStyle = '#fff';
        ctx.font = '12px Courier New';
        ctx.textAlign = 'center';
        ctx.fillText('⬆⬇⬅➡ Hetsika | SPACE Mitsambikina | 🖱 Kitiho Mipoaka', 300, 30);
        
        vovoka5.sariho(ctx);
    }
}));

lalao5.atombohy('input');

// ==========================================
// DEMO 6: Plugin Fable (Quests/XP/Inventory)
// ==========================================

const lalao6 = new Rakitrakatra2.Lalao(600, 300, {
    canvas: document.getElementById('canvas6')
});

let fable6, asaTanana;

lalao6.sehatra('fable', ({ fable }) => ({
    famoronana() {
        fable6 = fable;
        
        // Mamorona iraka
        asaTanana = fable6.iraka(
            'asa01',
            'Angony ny voankazo 5',
            [
                { label: 'Voankazo voaangona', ilaina: 5, manana: 0, vita: false }
            ],
            100
        );
        
        console.log('📋 Iraka noforonina:', asaTanana);
        
        // Henoy zava-mitranga
        fable6.reHetsika('irakaVita', (iraka) => {
            console.log('🎉 Iraka vita!', iraka.lohateny);
            Rakitrakatra2.Feo.sfx('power');
        });
        
        fable6.reHetsika('niveau', (haavo) => {
            console.log('⭐ Niakatra!', 'Niveau', haavo);
            Rakitrakatra2.Feo.sfx('power');
        });
    },
    sariho(ctx) {
        ctx.fillStyle = '#0a0a1a';
        ctx.fillRect(0, 0, 600, 300);
        
        // Asio raki-tsoratra eo amin'ny canvas
        ctx.fillStyle = '#ff1493';
        ctx.font = 'bold 18px Courier New';
        ctx.fillText('⭐ FABLE - Rafitra Kilalao', 150, 40);
        
        ctx.fillStyle = '#fff';
        ctx.font = '13px Courier New';
        ctx.fillText(`Haavo: ${fable6.haavo} | XP: ${fable6.xp}/${fable6.xpSeuil(fable6.haavo)}`, 30, 80);
        ctx.fillText(`Iraka: ${asaTanana.lohateny}`, 30, 100);
        ctx.fillText(`Fandrosoana: ${asaTanana.tanjona[0].manana}/${asaTanana.tanjona[0].ilaina} ${asaTanana.vita ? '✅ VITA!' : '⏳ Eo am-panamboarana...'}`, 30, 120);
        
        ctx.fillStyle = '#aaa';
        ctx.fillText('Kitapo: ' + fable6.kitapo.map(i => `${i.ID}(${i.isa})`).join(', ') || '(foana)', 30, 160);
        
        ctx.fillStyle = '#666';
        ctx.font = '11px Courier New';
        ctx.fillText('Jereo ny console (F12) hahitana ny zava-mitranga rehetra', 30, 200);
    }
}));

lalao6.atombohy('fable');

// Fonctions fanampiny
function manomeXP() {
    const xp = Math.floor(Math.random() * 50) + 10;
    fable6.omeoXp(xp);
    console.log(`⭐ +${xp} XP!`);
}

function vitaoIraka() {
    const fandrosoana = Math.floor(Math.random() * 2) + 1;
    fable6.fandrosoana('asa01', 0, fandrosoana);
    console.log(`📋 Fandrosoana +${fandrosoana} ho an'ny iraka!`);
}

function ampidiroZavatra() {
    const zavatra = [
        { ID: 'sabatra', anarana: 'Sabatra' },
        { ID: 'ampinga', anarana: 'Ampinga' },
        { ID: 'fanafody', anarana: 'Fanafody' },
        { ID: 'vola', anarana: 'Vola' }
    ];
    const z = zavatra[Math.floor(Math.random() * zavatra.length)];
    fable6.ampidiro(z);
    console.log('🎁 Nahazo:', z.anarana);
}
